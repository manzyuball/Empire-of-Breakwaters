# 軍閥への統合要求
country_event = {
	id = CHI_unification.1
	title = CHI_unification.1.t
	desc = CHI_unification.1.d
	picture = GFX_report_event_chinese_officers

	is_triggered_only = yes

	option = {
		name = CHI_unification.1.a
		ai_chance = { factor = 50 }
		CHI = { add_to_faction = ROOT }
		CHI = { annex_country = { target = ROOT transfer_troops = yes } }
	}
	option = {
		name = CHI_unification.1.b
		ai_chance = { factor = 50 }
		CHI = {
			create_wargoal = {
				type = take_state_focus
				target = ROOT
			}
		}
	}
}
