# 1936年選挙イベント
# =========================================================

# 選挙結果
country_event = {
    id = usa.10
    title = usa.10.t
    desc = usa.10.desc
    picture = GFX_report_event_usa_election

    is_triggered_only = yes

    option = {
        name = usa.10.a
        # Huey Longの勝利
        trigger = {
            OR = {
                check_variable = { USA_long_support > USA_mac_support }
                NOT = { has_variable = USA_mac_support }
            }
        }
        set_country_flag = USA_long_route
        custom_effect_tooltip = USA_long_route_tt
    }

    option = {
        name = usa.10.b
        # Douglas MacArthurの勝利
        trigger = {
            OR = {
                check_variable = { USA_mac_support > USA_long_support }
                NOT = { has_variable = USA_long_support }
            }
        }
        set_country_flag = USA_mac_route
        custom_effect_tooltip = USA_mac_route_tt
    }

    option = {
        name = usa.10.c
        # プレイヤー選択（同点の場合）
        trigger = {
            check_variable = { USA_long_support = USA_mac_support }
        }
        hidden_effect = {
            country_event = { id = usa.11 days = 1 }
        }
    }
}
