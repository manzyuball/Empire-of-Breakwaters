	id = usa.11
	title = usa.11.t
	desc = usa.11.d
	picture = GFX_report_event_usa_election_1936
	
	fire_only_once = yes
	trigger = {
		tag = USA
		date > 1936.11.3
	}
	mean_time_to_happen = { days = 1 }

	# A. 「民衆の声を聞け！」 (ロング)
	option = {
		name = usa.11.a
		trigger = {
			# Could add check for variable support, but user didn't specify strict requirement,
			# just branching.
		}
		
		# Set Huey Long as leader
		# Note: Huey Long must be defined in characters. Assuming he exists or vanilla.
		# Vanilla tag for Huey Long is often USA_huey_long.
		retire_country_leader = yes
		set_rule = { can_create_collaboration_government = yes } # Unlock plunder mechanics? (flavor)
		
		# Remove Hoover trait/ideas if needed, or they persist until focus tree removes them?
		# User said "economic reconstruction: forced seizure" etc. which sounds like focus tree.
		# For now, just change leader.
		
		create_country_leader = {
			name = "Huey Long"
			desc = "POLITICS_HUEY_LONG_DESC"
			picture = "GFX_portrait_USA_huey_long"
			expire = "1965.1.1"
			ideology = fascism_ideology # Or authoritarian/populist? User said 'Plunder'. Fascism fits 'Revanchist'.
			traits = {
				# dictator?
			}
		}
		set_politics = {
			ruling_party = fascism
			elections_allowed = no
		}
	}

	# B. 「秩序が必要だ。」 (マッカーサー)
	option = {
		name = usa.11.b
		
		# Set MacArthur as leader
		retire_country_leader = yes
		
		create_country_leader = {
			name = "Douglas MacArthur"
			desc = "POLITICS_DOUGLAS_MACARTHUR_DESC"
			picture = "GFX_portrait_USA_douglas_macarthur"
			expire = "1965.1.1"
			ideology = neutrality # Junta/Despotism
			traits = {
				# military_junta?
			}
		}
		set_politics = {
			ruling_party = neutrality
			elections_allowed = no
		}
	}
}
