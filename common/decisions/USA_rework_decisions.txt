	USA_state_industrial_investment = {

		allowed = {

		}
		available = {

			FROM = {

		}

			is_owned_and_controlled_by = USA

		cost = 50
		ai_will_do = { factor = 10 }

			FROM = {

				}
			}
			USA_add_factory_output_small = yes
			add_stability = 0.01
		}
	}

	# Decision 2: 緊急資源調査
	USA_emergency_resource_survey = {
		icon = GFX_decision_generic_raw_materials
		allowed = {
			tag = USA
		}
		available = {
			has_completed_focus = USA_trade_stabilization
			NOT = { has_country_flag = USA_resource_survey_done }
		}
		visible = {
			has_completed_focus = USA_hollow_state
		}
		fire_only_once = yes
		days_remove = 90
		cost = 75
		ai_will_do = { factor = 15 }
		complete_effect = {
			set_country_flag = USA_resource_survey_done
			custom_effect_tooltip = USA_resource_survey_start_tt
		}
		remove_effect = {
			random_owned_controlled_state = {
				limit = { has_resources = { steel > 5 } }
				add_resource = { type = steel amount = 5 }
			}
			random_owned_controlled_state = {
				limit = { has_resources = { oil > 3 } }
				add_resource = { type = oil amount = 3 }
			}
			add_to_variable = { USA_eco_resource_factor = 0.08 }
			USA_add_factory_output_small = yes
			custom_effect_tooltip = USA_resource_survey_complete_tt
		}
		modifier = {
			local_resources_factor = 0.05
			consumer_goods_factor = 0.02
		}
	}

	# Decision 3: 緊急労働力動員
	USA_emergency_labor_mobilization = {
		icon = GFX_decision_generic_manpower
		allowed = {
			tag = USA
		}
		available = {
			has_completed_focus = USA_social_stabilization
			OR = {
				has_idea = low_economic_mobilisation
				has_idea = partial_economic_mobilisation

			}
		}
		visible = {
			has_completed_focus = USA_hollow_state
		}
		fire_only_once = no
		days_remove = 120
		days_re_enable = 240
		cost = 100
		ai_will_do = { factor = 20 modifier = { factor = 2 has_war = yes } }
		complete_effect = {
			custom_effect_tooltip = USA_labor_mobilization_start_tt
		}
		remove_effect = {
			add_manpower = 50000
			USA_add_factory_output_medium = yes
			add_stability = -0.03
			custom_effect_tooltip = USA_labor_mobilization_end_tt
		}
		modifier = {
			production_speed_industrial_complex_factor = 0.15
			production_speed_arms_factory_factor = 0.15
