	# 中央工業地帯の浸透
	CHI_seize_central_zone = {
		icon = generic_industry
		available = {
			any_owned_state = { CHI_is_central_industrial_zone = yes }
			OR = {
				has_country_flag = CHI_influence_A_unlocked
				has_country_flag = CHI_influence_B_unlocked
				has_country_flag = CHI_influence_C_unlocked
			}
			NOT = { has_country_flag = CHI_route_determined }
		}
		visible = {
			OR = {
				has_country_flag = CHI_influence_A_unlocked
				has_country_flag = CHI_influence_B_unlocked
				has_country_flag = CHI_influence_C_unlocked
			}
			NOT = { has_country_flag = CHI_route_determined }
		}
		cost = 50
		days_remove = 60
		fire_only_once = no

		ai_will_do = { factor = 8 }

		complete_effect = {
			# 最大影響力の派閥がゾーンを浸透（+8）、他派閥は-3
			if = {
				limit = {
					check_variable = { var = CHI.influence_A value = CHI.influence_B compare = greater_than_or_equals }
					check_variable = { var = CHI.influence_A value = CHI.influence_C compare = greater_than_or_equals }
				}
				add_to_variable = { var = CHI.influence_A value = 8 }
				add_to_variable = { var = CHI.influence_B value = -3 }
				add_to_variable = { var = CHI.influence_C value = -3 }
			}
			else_if = {
				limit = {
					check_variable = { var = CHI.influence_B value = CHI.influence_C compare = greater_than_or_equals }
				}
				add_to_variable = { var = CHI.influence_B value = 8 }
				add_to_variable = { var = CHI.influence_A value = -3 }
				add_to_variable = { var = CHI.influence_C value = -3 }
			}
			else = {
				add_to_variable = { var = CHI.influence_C value = 8 }
				add_to_variable = { var = CHI.influence_A value = -3 }
				add_to_variable = { var = CHI.influence_B value = -3 }
			}
			country_event = { id = CHI_influence.21 }
		}
	}

		}

		complete_effect = {
			# 最大影響力の派閥に+5、安定度-3%（プロパガンダの社会的コスト）

			else_if = {
				limit = {
					check_variable = { var = CHI.influence_B value = CHI.influence_C compare = greater_than_or_equals }
				}
				add_to_variable = { var = CHI.influence_B value = 5 }
			}
			else = {
				add_to_variable = { var = CHI.influence_C value = 5 }
			}
			add_stability = -0.03

		}

		days_remove = 60
		days_re_enable = 60

		# 防衛中: 妨害工作を受けても影響力損失が政治的に緩和される
		modifier = {
			political_power_factor = -0.05
		}

		ai_will_do = { factor = 4 }

		remove_effect = {
			# 防衛期間終了: 最大勢力に小さなボーナス
			if = {
				limit = {
					check_variable = { var = CHI.influence_A value = CHI.influence_B compare = greater_than_or_equals }
					check_variable = { var = CHI.influence_A value = CHI.influence_C compare = greater_than_or_equals }
				}
				add_to_variable = { var = CHI.influence_A value = 3 }
			}
			else_if = {
				limit = {
					check_variable = { var = CHI.influence_B value = CHI.influence_C compare = greater_than_or_equals }
				}
				add_to_variable = { var = CHI.influence_B value = 3 }
			}
			else = {
				add_to_variable = { var = CHI.influence_C value = 3 }
			}
		}

	# 影響力判定→ルート確定
	CHI_check_influence_and_lock_route = {
		icon = generic_prepare_for_war
		available = {
			has_completed_focus = CHI_order_prepare
			has_completed_focus = CHI_paris_prepare
			has_completed_focus = CHI_moscow_prepare

			# 影響力変数が未初期化の場合は0をセット

			}
