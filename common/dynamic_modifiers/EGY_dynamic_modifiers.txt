# ============================================================================
# エジプト（EGY）動的修正子
# ============================================================================
# 「エジプト経済」：ディシジョン・イベント・NFで変動する複合経済システム
# 変数範囲：
#   EGY_eco_consumer_goods     (-0.40 ～ +0.75) ← 負が良い方向（消費財削減）
#   EGY_eco_stability          (-0.50 ～ +0.40)
#   EGY_eco_factory_output     (-0.30 ～ +0.25)
#   EGY_eco_dockyard_output    (-0.40 ～ +0.40)
#   EGY_eco_infra_speed        (-0.20 ～ +0.20)
#   EGY_eco_civ_speed          (-0.20 ～ +0.20)
#   EGY_eco_mil_speed          (-0.20 ～ +0.20)
#   EGY_eco_dock_speed         (-0.20 ～ +0.20)
# ============================================================================

EGY_economy_modifier = {
	icon = GFX_idea_generic_production_bonus
	enable = { always = yes }

	consumer_goods_factor                     = var:EGY_eco_consumer_goods
	stability_factor                          = var:EGY_eco_stability
	industrial_capacity_factory               = var:EGY_eco_factory_output
	industrial_capacity_dockyard              = var:EGY_eco_dockyard_output
	production_speed_infrastructure_factor    = var:EGY_eco_infra_speed
	production_speed_industrial_complex_factor = var:EGY_eco_civ_speed
	production_speed_arms_factory_factor      = var:EGY_eco_mil_speed
	production_speed_dockyard_factor          = var:EGY_eco_dock_speed
}

# ============================================================================
# 政治混乱度：イデオロギー変更・弾圧・改革で変動
# ============================================================================
#   EGY_pol_turmoil       (-0.30 ～ +0.10)  ← 負 = 不安定
#   EGY_pol_legitimacy    (-0.30 ～ +0.20)  ← 負 = PP減少
#   EGY_pol_repression    (-0.10 ～ +0.15)  ← 正 = 戦意増だが不安定化の種
#   EGY_pol_drift_defence (-0.30 ～ +0.30)
# ============================================================================

EGY_political_turmoil_modifier = {
	icon = GFX_idea_generic_political_power_bonus
	enable = { always = yes }

	stability_factor       = var:EGY_pol_turmoil
	political_power_factor = var:EGY_pol_legitimacy
	war_support_factor     = var:EGY_pol_repression
	drift_defence_factor   = var:EGY_pol_drift_defence
}

EGY_suez_canal_modifier = {
